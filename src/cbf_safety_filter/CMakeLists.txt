cmake_minimum_required(VERSION 3.16)
project(cbf_safety_filter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
set(CBF_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
foreach(pair osqp_0_6_3 qpoases_3_2_1 piqp_0_6_3)
  if(NOT EXISTS "${CBF_VENDOR}/${pair}/CMakeLists.txt")
    message(FATAL_ERROR "Missing pinned ${pair}; run scripts/fetch_teleop_dependencies.sh")
  endif()
endforeach()
set(PRINTING OFF CACHE BOOL "" FORCE)
set(PROFILING OFF CACHE BOOL "" FORCE)
set(CTRLC OFF CACHE BOOL "" FORCE)
set(ENABLE_MKL_PARDISO OFF CACHE BOOL "" FORCE)
set(UNITTESTS OFF CACHE BOOL "" FORCE)
set(OSQP_VERSION 0.6.3 CACHE STRING "" FORCE)
add_subdirectory("${CBF_VENDOR}/osqp_0_6_3" "${CMAKE_CURRENT_BINARY_DIR}/osqp" EXCLUDE_FROM_ALL)
# Build the LGPL dependency shared, allowing replacement without relinking the
# application. Source files are unmodified; no qpOASES console output in solves.
file(GLOB QPOASES_SOURCES "${CBF_VENDOR}/qpoases_3_2_1/src/*.cpp")
add_library(cbf_qpoases SHARED ${QPOASES_SOURCES})
target_include_directories(cbf_qpoases PUBLIC "$<BUILD_INTERFACE:${CBF_VENDOR}/qpoases_3_2_1/include>")
target_compile_definitions(cbf_qpoases PRIVATE __NO_COPYRIGHT__ __SUPPRESSANYOUTPUT__)
add_library(cbf_safety_filter SHARED src/configuration.cpp src/kinematics.cpp src/constraints.cpp src/qp_problem.cpp src/problem_snapshot.cpp src/solver_factory.cpp src/safety_filter.cpp src/recovery_projection.cpp
  src/solvers/osqp_solver.cpp src/solvers/qpoases_solver.cpp src/solvers/piqp_solver.cpp)
target_include_directories(cbf_safety_filter PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(cbf_safety_filter PRIVATE "${CBF_VENDOR}/piqp_0_6_3/include")
target_link_libraries(cbf_safety_filter PUBLIC Eigen3::Eigen PRIVATE yaml-cpp osqpstatic cbf_qpoases Threads::Threads)
set_target_properties(cbf_safety_filter PROPERTIES INSTALL_RPATH "$ORIGIN")
add_executable(cbf_validate src/validate.cpp)
target_link_libraries(cbf_validate cbf_safety_filter)
add_executable(cbf_replay src/replay.cpp)
target_link_libraries(cbf_replay cbf_safety_filter)
install(TARGETS cbf_safety_filter EXPORT export_cbf_safety_filter LIBRARY DESTINATION lib)
install(TARGETS cbf_qpoases EXPORT export_cbf_safety_filter LIBRARY DESTINATION lib)
install(TARGETS cbf_validate DESTINATION lib/${PROJECT_NAME})
install(TARGETS cbf_replay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config THIRD_PARTY_LICENSES DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_cbf_safety_filter HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  add_executable(benchmark_safety_filter test/benchmark_safety_filter.cpp)
  target_link_libraries(benchmark_safety_filter cbf_safety_filter)
  add_library(cbf_allocation_probe SHARED test/allocation_probe.cpp)
  ament_add_gtest(test_allocation_free test/test_allocation_free.cpp)
  target_link_libraries(test_allocation_free cbf_safety_filter dl)
  target_compile_definitions(test_allocation_free PRIVATE CBF_TEST_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/single_panda/cbf.yaml")
  set_tests_properties(test_allocation_free PROPERTIES ENVIRONMENT "LD_PRELOAD=$<TARGET_FILE:cbf_allocation_probe>")
  foreach(test_name test_bounded_recovery test_avoidance test_kinematics test_constraints test_qp test_filter test_problem_snapshot test_recorded_states)
    ament_add_gtest(${test_name} test/${test_name}.cpp)
    target_link_libraries(${test_name} cbf_safety_filter)
  endforeach()
  target_compile_definitions(test_avoidance PRIVATE CBF_TEST_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/single_panda/cbf.yaml")
  target_compile_definitions(test_recorded_states PRIVATE CBF_RECORDED_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../docs/incidents/2026-10-29/models")
  ament_add_gtest(test_configuration test/test_configuration.cpp)
  target_link_libraries(test_configuration cbf_safety_filter yaml-cpp)
  target_compile_definitions(test_configuration PRIVATE CBF_TEST_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/single_panda/cbf.yaml")
endif()
ament_package()
